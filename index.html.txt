const API_URL =
  'https://script.google.com/macros/s/AKfycbwsm6oqRxw26Ejq1L5XatKsDWiINGDCvgHD0a4qTIjoBf7vvnvPBmumdmtBlZSOxtPm/exec';


/* =========================================================
   GLOBAL VARIABLES
   ========================================================= */

let charts = {};

let comparisonData = [];

let jsonpCounter = 0;


/* =========================================================
   API CALL - JSONP
   ========================================================= */

function callAPI(action, params = {}) {

  return new Promise(function(resolve, reject) {

    const callbackName =
      '__hse_callback_' +
      Date.now() +
      '_' +
      (++jsonpCounter);


    const script =
      document.createElement('script');


    const queryParams =
      new URLSearchParams();


    queryParams.set(
      'action',
      action
    );


    queryParams.set(
      'callback',
      callbackName
    );


    Object.keys(params).forEach(function(key) {

      queryParams.set(
        key,
        params[key]
      );

    });


    script.src =
      API_URL +
      '?' +
      queryParams.toString();


    let completed = false;


    function cleanup() {

      delete window[callbackName];

      if (
        script.parentNode
      ) {

        script.parentNode.removeChild(
          script
        );

      }

    }


    window[callbackName] =
      function(data) {

        if (completed) {
          return;
        }

        completed = true;

        cleanup();


        if (
          data &&
          data.error
        ) {

          reject(
            new Error(
              data.message ||
              'API Error'
            )
          );

          return;

        }


        resolve(data);

      };


    script.onerror =
      function() {

        if (completed) {
          return;
        }

        completed = true;

        cleanup();


        reject(
          new Error(
            'Unable to connect to HSE Dashboard API.'
          )
        );

      };


    document.body.appendChild(
      script
    );

  });

}


/* =========================================================
   LOAD PROJECTS
   ========================================================= */

function loadProjects() {

  callAPI(
    'getProjects'
  )

  .then(function(projects) {

    const select =
      document.getElementById(
        'project'
      );


    select.innerHTML =
      '<option value="All Projects">' +
      'All Projects' +
      '</option>';


    if (
      Array.isArray(projects)
    ) {

      projects.forEach(
        function(project) {

          const option =
            document.createElement(
              'option'
            );


          option.value =
            project;


          option.textContent =
            project;


          select.appendChild(
            option
          );

        }
      );

    }


    loadDashboard();

  })


  .catch(function(error) {

    console.error(
      'Project loading error:',
      error
    );


    loadDashboard();

  });

}

   LOAD DASHBOARD
   ========================================================= */

function loadDashboard() {

  const year =
    document.getElementById(
      'year'
    ).value;


  const month =
    document.getElementById(
      'month'
    ).value;


  const project =
    document.getElementById(
      'project'
    ).value;


  callAPI(
    'getDashboardData',
    {

      year:
        year,

      month:
        month,

      project:
        project

    }
  )


  .then(function(data) {

    updateDashboard(
      data
    );

  })


  .catch(function(error) {

    console.error(
      'Dashboard loading error:',
      error
    );


    alert(
      'Error loading dashboard: ' +
      error.message
    );

  });


  loadProjectComparison();

}

/* =========================================================
   DRAW PROJECT COMPARISON
   ========================================================= */

function drawProjectComparison() {

  const canvas =
    document.getElementById(
      'projectComparisonChart'
    );


  if (!canvas) {
    return;
  }


  const metric =
    document.getElementById(
      'comparisonMetric'
    ).value;


  const labels =
    comparisonData.map(
      function(item) {

        return item.project;

      }
    );


  const values =
    comparisonData.map(
      function(item) {

        return Number(
          item[metric] || 0
        );

      }
    );


  if (
    charts.projectComparison
  ) {

    charts.projectComparison.destroy();

  }


  charts.projectComparison =
    new Chart(

      canvas,

      {

        type: 'bar',


        data: {

          labels:
            labels,


          datasets: [{

            label:
              getMetricLabel(
                metric
              ),


            data:
              values,


            borderWidth:
              1

          }]

        },


        options: {

          responsive:
            true,


          maintainAspectRatio:
            false,


          plugins: {

            legend: {

              display:
                false

            }

          },


          scales: {

            y: {

              beginAtZero:
                true

            }

          }

        }

      }

    );

}

/* =========================================================
   FORMATTERS
   ========================================================= */

function formatNumber(value) {

  const num =
    Number(
      value || 0
    );


  if (
    !isFinite(num)
  ) {

    return '0';

  }


  return num.toLocaleString(
    'en-US',
    {
      maximumFractionDigits:
        2
    }
  );

}


function formatPercent(value) {

  const num =
    Number(
      value || 0
    );


  if (
    !isFinite(num)
  ) {

    return '0%';

  }


  return num.toLocaleString(
    'en-US',
    {
      maximumFractionDigits:
        2
    }
  ) + '%';

}


/* =========================================================
   HTML ESCAPE
   ========================================================= */

function escapeHtml(value) {

  return String(
    value || ''
  )

    .replace(
      /&/g,
      '&amp;'
    )

    .replace(
      /</g,
      '&lt;'
    )

    .replace(
      />/g,
      '&gt;'
    )

    .replace(
      /"/g,
      '&quot;'
    )

    .replace(
      /'/g,
      '&#039;'
    );

}
